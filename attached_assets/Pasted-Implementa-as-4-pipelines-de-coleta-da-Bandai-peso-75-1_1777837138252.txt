═══ PARTE 2: Pipeline base Bandai compartilhada ═══

Cria lib/evidence/pipelines/bandai/shared.ts com lógica reutilizada pelas 4
pipelines (descobrir reports, parsear player.php, mapear aliases):

  1. discoverReports(eventTypePattern: RegExp) — fetch
     world.digimoncard.com/event/, parseia HTML procurando 
     `<a href="/report/...">`, retorna lista de URLs novos (não vistos
     antes — verifica via meta_archetype_evidences.url) que casam com 
     eventTypePattern.

  2. parseReportPage(reportUrl) — fetch report URL, extrai:
     - eventLabel (h1 do report ou title)
     - eventDate (do título/conteúdo, formato YYYY-MM-DD)
     - playerListUrl (link "View deck recipes of all participants")
     - topPlacements (Champion/Runner-Up/3rd se mencionados)

  3. parsePlayerListPage(playerListUrl) — fetch, parseia tabelas com
     headers "Player Name | Area | Deck", retorna:
     [{ playerName, area, deckName }, ...]

  4. mapDeckNameToArchetypeId(deckName, aliases) — normaliza (trim,
     case-insensitive), tenta match contra cada alias do dicionário.
     Se não casar, registra em unmatched_archetype_names (jsonb num arquivo
     temp) pra revisão manual. Retorna archetypeId ou null.

  5. aggregateAppearances(participants, aliases) — agrupa por archetypeId,
     conta appearances, calcula share_pct (count/total*100).

  6. importEventReport(sourceId, reportUrl) — combina tudo:
     a. parseReportPage(reportUrl)
     b. parsePlayerListPage(playerListUrl)
     c. mapeia deckNames pra archetypeIds
     d. agrega appearances
     e. pra cada arquetipo, upsertEvidence({
          gameId: "digimon",
          archetypeId,
          sourceId,
          eventLabel,
          eventDate,
          url: reportUrl,
          data: {
            appearances,
            share_pct,
            sample_size: appearances,
            placement: topPlacements[archetypeId] ?? null,
            won_event: champion?.archetypeId === archetypeId,
          },
          verified: true,                           // bandai = auto_verified
          verifiedBy: "bandai-auto-import",
          verifiedAt: NOW(),
        })

═══ PARTE 3: Fingerprint validator Bandai ═══

lib/evidence/pipelines/bandai/fingerprint.ts:

  export async function validateBandaiFingerprint(): Promise<FingerprintCheck> {
    const failures = [];
    
    // Test 1: página de eventos retorna HTML válido
    const eventPageRes = await fetch("https://world.digimoncard.com/event/", { 
      signal: AbortSignal.timeout(8000) 
    });
    if (!eventPageRes.ok) failures.push("Event page returned " + eventPageRes.status);
    const html = await eventPageRes.text();
    
    // Test 2: tem pelo menos 1 link de report
    if (!/<a[^>]+href="\/report\/[^"]+"/.test(html)) {
      failures.push("Nenhum link /report/* encontrado na página de eventos");
    }
    
    // Test 3: pega primeiro report e valida que tem link de player.php
    const reportMatch = html.match(/href="(\/report\/[^"]+)"/);
    if (reportMatch) {
      const reportRes = await fetch("https://world.digimoncard.com" + reportMatch[1]);
      const reportHtml = await reportRes.text();
      if (!/player\.php|deck recipes of all participants/i.test(reportHtml)) {
        failures.push("Report sample não tem link 'View deck recipes' — formato mudou");
      }
    } else {
      failures.push("Não foi possível pegar report sample pra validar formato");
    }
    
    return { ok: failures.length === 0, failures };
  }

═══ PARTE 4: 4 pipelines específicas ═══

Cada uma é uma classe simples implementando EvidencePipeline. Diferença é só
o eventTypePattern e sourceId:

lib/evidence/pipelines/bandai/worlds-final.ts:

  export class BandaiWorldsFinalPipeline implements EvidencePipeline {
    sourceId = "bandai-worlds-final";
    
    async validateFingerprint() { return validateBandaiFingerprint(); }
    
    async import(): Promise<PipelineRun> {
      const reports = await discoverReports(/world.*final/i);
      let itemsImported = 0;
      const archetypesUpdated = new Set<string>();
      const warnings: string[] = [];
      
      for (const url of reports) {
        try {
          const result = await importEventReport(this.sourceId, url);
          itemsImported += result.archetypeCount;
          result.archetypeIds.forEach(id => archetypesUpdated.add(id));
        } catch (err) {
          warnings.push(`Falha em ${url}: ${(err as Error).message}`);
        }
      }
      
      return { itemsImported, archetypesUpdated: [...archetypesUpdated], warnings };
    }
  }
