  export class DigimonMetaPipeline implements EvidencePipeline {
    sourceId = "digimonmeta-review";
    
    async validateFingerprint(): Promise<FingerprintCheck> {
      const failures = [];
      const res = await fetch("https://digimonmeta.com/wp-json/wp/v2/categories?slug=deckreview", {
        headers: { Accept: "application/json" },
        signal: AbortSignal.timeout(8000),
      });
      if (!res.ok) failures.push("WP REST API returned " + res.status);
      const cats = await res.json();
      if (!Array.isArray(cats) || cats.length === 0 || cats[0].id !== 280) {
        failures.push("Categoria 'deckreview' não encontrada com ID 280 — pode ter sido renomeada/deletada");
      }
      return { ok: failures.length === 0, failures };
    }
    
    async import(): Promise<PipelineRun> {
      const game = await getGame("digimon");
      const aliases = game.config.archetype_aliases;
      
      // Pega posts da última coleta
      const lastRun = await getLastSuccessfulRun("digimonmeta-review");
      const since = lastRun?.detectedAt ?? new Date("2024-01-01");
      const sinceISO = since.toISOString();
      
      const url = `https://digimonmeta.com/wp-json/wp/v2/posts?categories=280&per_page=20&after=${sinceISO}&_fields=id,date,title,content,slug,link`;
      const res = await fetch(url, { signal: AbortSignal.timeout(10000) });
      const posts = await res.json();
      
      let itemsImported = 0;
      const archetypesUpdated: string[] = [];
      const warnings: string[] = [];
      
      for (const post of posts) {
        const titleText = post.title.rendered;
        const contentHtml = post.content.rendered;
        
        // Heurística: detecta archetypeId pelo título OU pelo content
        const archetypeId = detectArchetypeFromText(titleText, contentHtml, aliases);
        if (!archetypeId) {
          warnings.push(`Post "${titleText.slice(0, 50)}" não mapeado pra arquetipo`);
          continue;
        }
        
        // Conta cards mencionados pra dar uma indicação de "profundidade"
        const cardMentions = (contentHtml.match(/[BSE][TPX]\d+-\d+/g) ?? []).length;
        
        await upsertEvidence({
          gameId: "digimon",
          archetypeId,
          sourceId: "digimonmeta-review",
          eventLabel: titleText.replace(/<[^>]+>/g, "").slice(0, 200),
          eventDate: new Date(post.date),
          url: post.link,
          data: {
            type: "review_article",
            cards_mentioned: cardMentions,
            author_slug: post.slug.split("-")[0] ?? null,
          },
          verified: true,  // editorial é auto_verified
          verifiedBy: "digimonmeta-auto-import",
        });
        
        itemsImported++;
        archetypesUpdated.push(archetypeId);
      }
      
      return { itemsImported, archetypesUpdated, warnings };
    }
  }

  function detectArchetypeFromText(title, contentHtml, aliases) {
    // 1. Tenta match no título primeiro (alta confiança)
    for (const [archetypeId, names] of Object.entries(aliases)) {
      for (const name of names) {
        if (title.toLowerCase().includes(name.toLowerCase())) {
          return archetypeId;
        }
      }
    }
    // 2. Fallback: conta menções de cards no content e tenta inferir
    // (pula por enquanto — só usa título)
    return null;
  }

  export class DigimonCardIoPipeline implements EvidencePipeline {
    sourceId = "digimoncard-io";
    
    async validateFingerprint(): Promise<FingerprintCheck> {
      const failures = [];
      // Tenta endpoint conhecido (cards) pra confirmar API up
      const res = await fetch("https://digimoncard.io/api-public/search?card=BT13-040", {
        headers: { Accept: "application/json" },
        signal: AbortSignal.timeout(8000),
      });
      if (!res.ok) failures.push("digimoncard.io card API returned " + res.status);
      
      // Tenta endpoint de decks
      const decksRes = await fetch("https://digimoncard.io/category/meta-decks", {
        signal: AbortSignal.timeout(8000),
      });
      if (!decksRes.ok) failures.push("digimoncard.io meta-decks page returned " + decksRes.status);
      
      return { ok: failures.length === 0, failures };
    }
    
    async import(): Promise<PipelineRun> {
      const game = await getGame("digimon");
      const evSource = game.config.evidence_sources.find(s => s.id === "digimoncard-io");
      const minSample = evSource?.min_sample_size ?? 200;
      const aliases = game.config.archetype_aliases;
      
      // PRIMEIRA TENTATIVA: endpoint JSON
      // Tenta /api-public/decks ou /api-public/meta — se não existir, fallback pra HTML scraping
      let archetypes = [];
      try {
        const apiRes = await fetch("https://digimoncard.io/api-public/decks", {
          headers: { Accept: "application/json" },
          signal: AbortSignal.timeout(10000),
        });
        if (apiRes.ok) {
          const data = await apiRes.json();
          // Adapta formato — esperamos array com { name, deck_count, win_rate, ... }
          archetypes = parseDigimonCardIoApi(data);
        }
      } catch {
        // Fallback pra HTML scraping
      }
      
      if (archetypes.length === 0) {
        // Fallback: scraping HTML de /category/meta-decks
        const htmlRes = await fetch("https://digimoncard.io/category/meta-decks");
        const html = await htmlRes.text();
        archetypes = parseDigimonCardIoHtml(html);
      }
      
      let itemsImported = 0;
      const archetypesUpdated: string[] = [];
      const warnings: string[] = [];
      
      for (const arch of archetypes) {
        if ((arch.sample_size ?? 0) < minSample) continue;
        
        const archetypeId = mapDeckNameToArchetypeId(arch.name, aliases);
        if (!archetypeId) {
          warnings.push(`DigimonCard.io: "${arch.name}" não mapeado`);
          continue;
        }
        
        await upsertEvidence({
          gameId: "digimon",
          archetypeId,
          sourceId: "digimoncard-io",
          eventLabel: `DigimonCard.io agregado ${formatWeek(new Date())}`,
          eventDate: new Date(),
          url: arch.url ?? "https://digimoncard.io/category/meta-decks",
          data: {
            sample_size: arch.sample_size,
            win_rate: arch.win_rate ?? null,
            share_pct: arch.share_pct ?? null,
            uploads_count: arch.uploads_count ?? null,
          },
          verified: false,  // self-reported nunca é auto_verified
        });
        
        itemsImported++;
        archetypesUpdated.push(archetypeId);
      }
      
      return { itemsImported, archetypesUpdated, warnings };
    }
  }

  function parseDigimonCardIoApi(data) {
    // Adapta — schema deles é desconhecido. Quando rodar a primeira vez,
    // loga estrutura recebida e ajusta.
    if (!Array.isArray(data)) return [];
    return data.map(d => ({
      name: d.name ?? d.archetype ?? d.deck_name,
      sample_size: d.deck_count ?? d.uploads_count ?? d.sample,
      win_rate: d.win_rate ?? d.win_pct ?? null,
      share_pct: d.share ?? d.share_pct ?? null,
      uploads_count: d.uploads_count ?? d.deck_count ?? null,
      url: d.url ?? null,
    })).filter(a => a.name);
  }
